import jwt from "jsonwebtoken";
import { JWT_SECRET } from "../config/jwt.js";

export default function auth(req, res, next) {
  const authHeader = req.headers.authorization;

  if (!authHeader)
    return res.status(401).json({ message: "No token" });

  const token = authHeader.split(" ")[1];

  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    req.user = decoded;
    next();
  } catch {
    res.status(401).json({ message: "Invalid token" });
  }
}

export default class User {
  static async create({ name, email, password }) {
    const hash = await bcrypt.hash(password, 10);

    const [result] = await pool.query(
      "INSERT INTO members (name,email,password,role) VALUES (?,?,?,?)",
      [name, email, hash, "member"]
    );

    return { id: result.insertId, name, email, role: "member" };
  }

  static async findByEmail(email) {
    const [rows] = await pool.query(
      "SELECT * FROM members WHERE email=?",
      [email]
    );
    return rows[0];
  }
}

const auth = (req, res, next) => {

import express from "express";
import bcrypt from "bcryptjs";
import jwt from "jsonwebtoken";
import { pool } from "../config/db.js";

const router = express.Router();
const JWT_SECRET = "MTCLUB_SECRET";

// ✅ تسجيل حساب جديد
router.post("/register", async (req, res) => {
  const { name, email, password } = req.body;

  try {
    const [existing] = await pool.query(
      "SELECT * FROM members WHERE email = ?",
      [email]
    );

    if (existing.length > 0) {
      return res.status(400).json({ message: "Email already exists" });
    }

    const hashedPassword = await bcrypt.hash(password, 10);

    await pool.query(
      "INSERT INTO members (name, email, password) VALUES (?, ?, ?)",
      [name, email, hashedPassword]
    );

    res.json({ message: "Registered successfully" });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
});

// 🔐 تسجيل دخول
router.post("/login", async (req, res) => {
  const { email, password } = req.body;

  try {
    const [rows] = await pool.query(
      "SELECT * FROM members WHERE email = ?",
      [email]
    );

    if (rows.length === 0)
      return res.status(401).json({ message: "Invalid credentials" });

    const user = rows[0];
    const match = await bcrypt.compare(password, user.password);

    if (!match)
      return res.status(401).json({ message: "Invalid credentials" });

    const token = jwt.sign(
      { id: user.id, role: user.role },
      JWT_SECRET,
      { expiresIn: "7d" }
    );

    res.json({
      id: user.id,
      name: user.name,
      email: user.email,
      role: user.role,
      token,
    });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
});

export default router;

import { useState, useEffect } from "react";

export default function AuthModal({ onClose, onLogin }) {

  const [mode, setMode] = useState("signin");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {

    document.body.style.overflow = "hidden";

    return () => {
      document.body.style.overflow = "auto";
    };

  }, []);

  const handleSubmit = async (e) => {

    e.preventDefault();
    setLoading(true);
    setError("");

    const form = new FormData(e.target);

    const club_id = form.get("club_id");
    const name = form.get("name");
    const email = form.get("email");
    const password = form.get("password");
    const confirm = form.get("confirm");

    try {

      if (mode === "signup") {

        if (password !== confirm) {

          setError("Passwords do not match");
          setLoading(false);
          return;

        }

        const res = await fetch("http://localhost:5001/api/auth/register", {

          method: "POST",

          headers: {
            "Content-Type": "application/json"
          },

          body: JSON.stringify({
            club_id,
            name,
            email,
            password
          })

        });

        const data = await res.json();

        if (!res.ok) {

          setError(data.error || "Registration failed");
          setLoading(false);
          return;

        }

        setMode("signin");
        setLoading(false);
        return;

      }

      const res = await fetch("http://localhost:5001/api/auth/login", {

        method: "POST",

        headers: {
          "Content-Type": "application/json"
        },

        body: JSON.stringify({
          email,
          password
        })

      });

      const data = await res.json();

      if (!res.ok) {

        setError(data.error || "Login failed");
        setLoading(false);
        return;

      }

      localStorage.setItem("user", JSON.stringify(data));

      onLogin(data);

      onClose?.();

    } catch (err) {

      setError("Server error");

    }

    setLoading(false);

  };

  return (

    <div className="fixed inset-0 flex items-center justify-center z-50">

            alt="MT Logo"
            className="w-20 h-20 object-contain"
          />

        </div>

        <h2 className="text-2xl font-semibold mb-6 text-center text-white">

          {mode === "signin" ? "Sign In" : "Sign Up"}

import { createContext, useState } from "react";

export const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  // استرجاع بيانات المستخدم من localStorage لو موجودة
  const [user, setUser] = useState(() => {
    const saved = localStorage.getItem("user"); // غيرت الاسم من "members" لـ "user"
    return saved ? JSON.parse(saved) : null;
  });

  const [showAuth, setShowAuth] = useState(false);

  const login = (userData) => {
    setUser(userData);
    localStorage.setItem("user", JSON.stringify(userData)); // نفس المفتاح هنا
  };

  const logout = () => {
    setUser(null);
    localStorage.removeItem("user");
  };

  const openAuth = () => setShowAuth(true);
  const closeAuth = () => setShowAuth(false);

  return (
    <AuthContext.Provider value={{ user, login, logout, showAuth, openAuth, closeAuth }}>
      {children}
    </AuthContext.Provider>
  );
};

export const UserContext = createContext();

export const UserProvider = ({ children }) => {
  const [user, setUser] = useState(null);

  useEffect(() => {
    const savedUser = localStorage.getItem("user");
    if (savedUser) setUser(JSON.parse(savedUser));
  }, []);

  const logout = () => {
    setUser(null);
    localStorage.removeItem("user");
  };

  return (
    <UserContext.Provider value={{ user, setUser, logout }}>
      {children}
    </UserContext.Provider>
  );
};

export default function Attendance({ user }) {
  const [attendance, setAttendance] = useState([]);
  const [search, setSearch] = useState("");

  // جلب بيانات الحضور
  useEffect(() => {
    axios
      .get("/api/attendance", {
        headers: { Authorization: `Bearer ${user.token}` },
      })
      .then((res) => setAttendance(res.data))
      .catch((err) => console.log(err));
  }, [user]);

  // فلترة البحث
  const filtered = attendance.filter(
    (a) =>
      a.memberName.toLowerCase().includes(search.toLowerCase()) ||
      a.eventTitle.toLowerCase().includes(search.toLowerCase())
  );

  // قفل الحدث
  const lockEvent = (eventId) => {
    axios
      .post(
        `/api/events/${eventId}/lock`,
        {},
        { headers: { Authorization: `Bearer ${user.token}` } }
      )
      .then(() => alert("Event Locked!"))
      .catch((err) => console.log(err));
  };

  // تصدير Excel / PDF (مبسّط)
  const exportCSV = () => {
    const csv = [
      ["Member Name", "Event Title", "Date", "Time"],
      ...attendance.map((a) => [a.memberName, a.eventTitle, a.date, a.time]),
    ]
      .map((e) => e.join(","))
      .join("\n");
    const blob = new Blob([csv], { type: "text/csv" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = "attendance.csv";
    link.click();
  };

  return (
    <div className="p-8 max-w-5xl mx-auto">
      <h2 className="text-2xl font-bold mb-4">Attendance Management</h2>

      <div className="flex gap-4 mb-4">
        <input

        <button
          onClick={exportCSV}
          className="bg-green-500 text-white px-4 py-2 rounded"
        >
          Export CSV
        </button>
      </div>

      <table className="min-w-full border">
        <thead>
          <tr className="bg-gray-200">
            <th className="p-2">Member Name</th>
            <th className="p-2">Event Title</th>
            <th className="p-2">Date</th>
            <th className="p-2">Time</th>
            <th className="p-2">Actions</th>
          </tr>
        </thead>
        <tbody>
          {filtered.map((a) => (
            <tr key={a.id} className="border-t">
              <td className="p-2">{a.memberName}</td>
              <td className="p-2">{a.eventTitle}</td>

import Header from "../components/Header";
import { useEffect, useState } from "react";
import { Link } from "react-router-dom";
import AuthModal from "../components/authmodal";

export default function Events({  user: propUser, setUser, onLogout, onLoaded  }) {
  const [events, setEvents] = useState([]);
    const [showAuthModal, setShowAuthModal] = useState(false);
  const [showModal, setShowModal] = useState(false);
  const [user, setLocalUser] = useState(propUser || null);
  const [newEvent, setNewEvent] = useState({ title: "", date: "", image: "" });

  // جلب الأيفنتات
  useEffect(() => {
    if (!user?.token) return;

    fetch("http://localhost:5001/api/events", {
      headers: {
        "Authorization": `Bearer ${user.token}`
      }
    })
      .then(res => res.json())
      .then(data => setEvents(Array.isArray(data) ? data : []))
      .catch(err => console.log(err));
  }, [user]);
  useEffect(() => {
    if (!propUser) {
      const savedUser = localStorage.getItem("user");
      if (savedUser) setLocalUser(JSON.parse(savedUser));
    }
  }, [propUser]);

  // إضافة حدث جديد
  const handleAddEvent = async () => {
    if (!newEvent.title || !newEvent.date || !newEvent.image) return;

    try {
      const res = await fetch("http://localhost:5001/api/events", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${user.token}`
        },
        body: JSON.stringify(newEvent)
      });

      if (!res.ok) throw new Error("Failed to add event");

      const data = await res.json();
      setEvents([...events, { ...newEvent, id: data.id }]);
      setNewEvent({ title: "", date: "", image: "" });
      setShowModal(false);
    } catch (err) {
      console.log(err);
    }
  };

  if (onLoaded) onLoaded();

  return (
    <main className="relative w-full h-screen overflow-y-auto overflow-x-hidden">

import React, { useState, useContext } from "react";
import { AuthContext } from "../context/AuthContext";
import QrReader from "react-qr-reader";

export default function ScanAttendance() {
  const { user } = useContext(AuthContext);
  const [manualCode, setManualCode] = useState("");
  const [message, setMessage] = useState("");

  const handleScan = async (data) => {
    if (data) await registerAttendance(data);
  };

  const handleError = (err) => console.error(err);

  const registerAttendance = async (data) => {
    let eventData;
    try {
      eventData = typeof data === "string" ? JSON.parse(data) : data;
    } catch {
      setMessage("Invalid QR code");
      return;
    }

    const location = await new Promise((resolve) => {
      navigator.geolocation.getCurrentPosition(
        pos => resolve(`${pos.coords.latitude},${pos.coords.longitude}`),
        () => resolve("Unknown")
      );
    });

    try {
      const res = await fetch("http://localhost:5001/api/attendance", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${user.token}`
        },
        body: JSON.stringify({
          memberId: user.id,
          eventId: eventData.eventId,
          location
        })
      });
      if (!res.ok) throw new Error("Failed to register attendance");
      setMessage("Attendance registered successfully!");
    } catch (err) {
      console.error(err);
      setMessage("Error registering attendance");
    }
  };

  const handleManualSubmit = (e) => {
    e.preventDefault();
    registerAttendance(manualCode);
  };

  return (
    <div className="p-6">
      <h1 className="text-3xl font-bold mb-4">Scan Attendance</h1>
      <div className="mb-4">
        <QrReader
          delay={300}
          onError={handleError}
          onScan={handleScan}
          style={{ width: "100%" }}
        />
      </div>

      <form onSubmit={handleManualSubmit} className="mb-4">
        <input
          type="text"
          value={manualCode}
          onChange={(e) => setManualCode(e.target.value)}
          placeholder="Enter QR code manually"
          className="border p-2 rounded w-full mb-2"
        />
        <button type="submit" className="px-4 py-2 bg-cyan-500 text-black rounded hover:bg-cyan-600">
          Submit
        </button>
      </form>

      {message && <p className="text-green-500">{message}</p>}
    </div>
  );
}

import jwt from "jsonwebtoken";
import { JWT_SECRET } from "../config/jwt.js";

export default function auth(req, res, next) {
  const authHeader = req.headers.authorization;

  if (!authHeader)
    return res.status(401).json({ message: "No token" });

  const token = authHeader.split(" ")[1];

  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    req.user = decoded;
    next();
  } catch {
    res.status(401).json({ message: "Invalid token" });
  }
}

router.post("/register", async (req, res) => {
  const { name, email, password } = req.body;

  try {
    const [existing] = await pool.query(
      "SELECT * FROM members WHERE email = ?",
      [email]
    );

    if (existing.length > 0) {
      return res.status(400).json({ message: "Email already exists" });
    }

    const hashedPassword = await bcrypt.hash(password, 10);

    await pool.query(
      "INSERT INTO members (name, email, password) VALUES (?, ?, ?)",
      [name, email, hashedPassword]
    );

    res.json({ message: "Registered successfully" });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
});

// 🔐 تسجيل دخول
router.post("/login", async (req, res) => {
  const { email, password } = req.body;

  try {
    const [rows] = await pool.query(
      "SELECT * FROM members WHERE email = ?",
      [email]
    );

    if (rows.length === 0)
      return res.status(401).json({ message: "Invalid credentials" });

    const user = rows[0];
    const match = await bcrypt.compare(password, user.password);

    if (!match)
      return res.status(401).json({ message: "Invalid credentials" });

    const token = jwt.sign(
      { id: user.id, role: user.role },
      JWT_SECRET,
      { expiresIn: "7d" }
    );

    res.json({
      id: user.id,
      name: user.name,
      email: user.email,
      role: user.role,
      token,
    });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
});

export default router;

    const name = form.get("name");
    const email = form.get("email");
    const password = form.get("password");
    const confirm = form.get("confirm");

          setError("Passwords do not match");
          setLoading(false);
          return;

import { createContext, useState } from "react";

export const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  // استرجاع بيانات المستخدم من localStorage لو موجودة
  const [user, setUser] = useState(() => {
    const saved = localStorage.getItem("user"); // غيرت الاسم من "members" لـ "user"
    return saved ? JSON.parse(saved) : null;
  });

  const [showAuth, setShowAuth] = useState(false);

  const login = (userData) => {
    setUser(userData);
    localStorage.setItem("user", JSON.stringify(userData)); // نفس المفتاح هنا
  };

  const logout = () => {
    setUser(null);
    localStorage.removeItem("user");
  };

  const openAuth = () => setShowAuth(true);
  const closeAuth = () => setShowAuth(false);

  return (
    <AuthContext.Provider value={{ user, login, logout, showAuth, openAuth, closeAuth }}>
      {children}
    </AuthContext.Provider>
  );
};

      .catch((err) => console.log(err));
  }, [user]);

  // فلترة البحث
  const filtered = attendance.filter(
    (a) =>
      a.memberName.toLowerCase().includes(search.toLowerCase()) ||
      a.eventTitle.toLowerCase().includes(search.toLowerCase())
  );

  // قفل الحدث
  const lockEvent = (eventId) => {
    axios
      .post(
        `/api/events/${eventId}/lock`,
        {},
        { headers: { Authorization: `Bearer ${user.token}` } }
      )
      .then(() => alert("Event Locked!"))
      .catch((err) => console.log(err));
  };

  // تصدير Excel / PDF (مبسّط)

    link.download = "attendance.csv";
    link.click();
  };

  return (
    <div className="p-8 max-w-5xl mx-auto">
      <h2 className="text-2xl font-bold mb-4">Attendance Management</h2>

export default function Events({  user: propUser, setUser, onLogout, onLoaded  }) {
  const [events, setEvents] = useState([]);
    const [showAuthModal, setShowAuthModal] = useState(false);
  const [showModal, setShowModal] = useState(false);
  const [user, setLocalUser] = useState(propUser || null);
  const [newEvent, setNewEvent] = useState({ title: "", date: "", image: "" });

  // جلب الأيفنتات
  useEffect(() => {
    if (!user?.token) return;

    fetch("http://localhost:5001/api/events", {
      headers: {
        "Authorization": `Bearer ${user.token}`
      }
    })
      .then(res => res.json())
      .then(data => setEvents(Array.isArray(data) ? data : []))
      .catch(err => console.log(err));
  }, [user]);
  useEffect(() => {
    if (!propUser) {
      const savedUser = localStorage.getItem("user");
      if (savedUser) setLocalUser(JSON.parse(savedUser));
    }
  }, [propUser]);

  // إضافة حدث جديد
  const handleAddEvent = async () => {
    if (!newEvent.title || !newEvent.date || !newEvent.image) return;

    try {
      const res = await fetch("http://localhost:5001/api/events", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${user.token}`
        },
        body: JSON.stringify(newEvent)
      });

      if (!res.ok) throw new Error("Failed to add event");

      const data = await res.json();
      setEvents([...events, { ...newEvent, id: data.id }]);
      setNewEvent({ title: "", date: "", image: "" });
      setShowModal(false);
    } catch (err) {
      console.log(err);
    }
  };

  if (onLoaded) onLoaded();

import React, { useState, useContext } from "react";
import { AuthContext } from "../context/AuthContext";
import QrReader from "react-qr-reader";

export default function ScanAttendance() {
  const { user } = useContext(AuthContext);
  const [manualCode, setManualCode] = useState("");
  const [message, setMessage] = useState("");

  const handleScan = async (data) => {
    if (data) await registerAttendance(data);
  };

  const handleError = (err) => console.error(err);

  const registerAttendance = async (data) => {
    let eventData;
    try {
      eventData = typeof data === "string" ? JSON.parse(data) : data;
    } catch {
      setMessage("Invalid QR code");
      return;
    }

    const location = await new Promise((resolve) => {
      navigator.geolocation.getCurrentPosition(
        pos => resolve(`${pos.coords.latitude},${pos.coords.longitude}`),
        () => resolve("Unknown")
      );
    });

    try {
      const res = await fetch("http://localhost:5001/api/attendance", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${user.token}`
        },
        body: JSON.stringify({
          memberId: user.id,
          eventId: eventData.eventId,
          location
        })
      });
      if (!res.ok) throw new Error("Failed to register attendance");
      setMessage("Attendance registered successfully!");
    } catch (err) {
      console.error(err);
      setMessage("Error registering attendance");
    }
  };

  const handleManualSubmit = (e) => {
    e.preventDefault();
    registerAttendance(manualCode);
  };

  return (
    <div className="p-6">
      <h1 className="text-3xl font-bold mb-4">Scan Attendance</h1>
      <div className="mb-4">
        <QrReader
          delay={300}
          onError={handleError}
          onScan={handleScan}
          style={{ width: "100%" }}
        />
      </div>

      <form onSubmit={handleManualSubmit} className="mb-4">
        <input
          type="text"
          value={manualCode}
          onChange={(e) => setManualCode(e.target.value)}
          placeholder="Enter QR code manually"
          className="border p-2 rounded w-full mb-2"
        />
        <button type="submit" className="px-4 py-2 bg-cyan-500 text-black rounded hover:bg-cyan-600">
          Submit
        </button>
      </form>

      {message && <p className="text-green-500">{message}</p>}
    </div>
  );
}

const API = "http://localhost:5001";

export async function registerUser(data) {

  const res = await fetch(API + "/register", {
    method: "POST",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify(data)
  });

  return res.json();
}

export async function loginUser(data) {

  const res = await fetch(API + "/login", {
    method: "POST",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify(data)
  });

  return res.json();
}

import { useState, useContext } from "react";
import { Routes, Route } from "react-router-dom";

import Home from "./pages/Home";
import Events from "./pages/Events";
import Profile from "./pages/Profile";
import TeamPage from "./pages/Team";
import Gallery from "./pages/Gallery";
import AuthModal from "./components/authmodal";
import { AuthProvider, AuthContext } from "./context/AuthContext";
import AdminPanel from "./pages/AdminPanel";
import Attendance from "./pages/Attendance";

function AppContent() {
  const { user, login, logout } = useContext(AuthContext);
  const [showAuth, setShowAuth] = useState(false);

  const openAuth = () => setShowAuth(true);
  const closeAuth = () => setShowAuth(false);

  return (
    <>
      <Routes>
        <Route path="/" element={<Home user={user} onLogout={logout} onLoginOpen={openAuth} />} />
        <Route path="/gallery" element={<Gallery user={user} onLogout={logout} onLoginOpen={openAuth} />} />
        <Route path="/events" element={<Events user={user} onLogout={logout} onLoginOpen={openAuth} />} />
        <Route path="/profile" element={<Profile user={user} onLogout={logout} onLoginOpen={openAuth} />} />
        <Route path="/team" element={<TeamPage user={user} onLogout={logout} onLoginOpen={openAuth} />} />
        <Route path="/admin" element={<AdminPanel user={user} onLogout={logout} onLoginOpen={openAuth} />} />
        <Route path="/attendance" element={<Attendance user={user} onLogout={logout} onLoginOpen={openAuth} />} />
      </Routes>

      {showAuth && (
        <AuthModal
          onClose={closeAuth}
          onLogin={(data) => {
            login(data);
            closeAuth();
          }}
        />
      )}
    </>
  );
}

export default function App() {
  return (
    <AuthProvider>
      <AppContent />
    </AuthProvider>
  );
}
